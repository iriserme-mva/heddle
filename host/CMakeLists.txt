# JUCE host layer (opt-in: -DHEDDLE_BUILD_HOST=ON)
#
# Fetches JUCE 8.0.4 (pinned by SHA256), builds the test VST3 plugin and
# the heddle_smoke CLI (JuceBackend + heddle_core). Core-only builds never
# touch JUCE, so the CI core job stays light.

include(FetchContent)
FetchContent_Declare(juce
    URL https://github.com/juce-framework/JUCE/archive/refs/tags/8.0.4.tar.gz
    URL_HASH SHA256=0effe9823a6fd8a4f98e5c08be7d308350e417b2d15dcf34011e778fefbbf005)
FetchContent_MakeAvailable(juce)

# JUCE's top-level CMakeLists already calls juce_add_modules(ALIAS_NAMESPACE juce)
# — do not call it again.

# Version header for the CLIs, stamped from the root project version.
configure_file(src/heddle_version.hpp.in heddle_version.hpp @ONLY)

add_subdirectory(test_plugin)
add_subdirectory(heddle_bridge)

# ---- smoke CLI -------------------------------------------------------------
add_executable(heddle_smoke
    src/smoke.cpp
    src/JuceBackend.cpp
)
target_link_libraries(heddle_smoke
    PRIVATE
        heddle_core
        juce::juce_core
        juce::juce_events
        juce::juce_audio_processors
)
# NOTE: we do NOT link juce::juce_audio_plugin_client into the smoke CLI.
# That is the *plugin-side* module; linking it would compile a second copy of
# the VST3 SDK base into the host executable, which conflicts (symbol
# interposition / ODR on the shared FUnknown/IPluginFactory base) with the SDK
# already compiled into the loaded .vst3 plugin and segfaults at instantiate
# time. JUCE's own AudioPluginHost links only juce_audio_processors (host side)
# for exactly this reason. JuceBackend.cpp / smoke.cpp use no client API.
# Headless host: disable the web browser so the juce_gui_extra module sources
# (compiled into this target) don't drag in GTK (Linux/BSD) or WebKit (macOS).
# Matches the way JUCE's own AudioPluginHost is configured.
target_compile_definitions(heddle_smoke PRIVATE JUCE_WEB_BROWSER=0)
# No networking: JUCE_USE_CURL defaults to 1 and #includes <curl/curl.h> on
# Linux/BSD, where CI runners lack libcurl dev headers. See test_plugin.
target_compile_definitions(heddle_smoke PRIVATE JUCE_USE_CURL=0)
# VST3 hosting: the AudioPluginFormatManager only registers the VST3 format
# when JUCE_PLUGINHOST_VST3=1 (per-target, like JUCE_USE_CURL). Without it
# the manager has zero formats and load() fails with "no plugin format can
# load this file". JUCE's own AudioPluginHost is configured this way.
target_compile_definitions(heddle_smoke PRIVATE JUCE_PLUGINHOST_VST3=1)
if(WIN32)
    target_link_libraries(heddle_smoke PRIVATE ws2_32 winmm)
endif()

# ---- render CLI (offline audio->audio) ------------------------------------
# heddle-render <input.wav> <chain.json> <output.wav> [--bits 16|24|32]
#
# Same host-side module set as the smoke CLI, plus juce_audio_formats for
# the WAV reader/writer. Re-uses JuceBackend.cpp so the renderer exercises
# exactly the same load/process path (including the param-push fix) that
# the live app uses.
add_executable(heddle-render
    src/render.cpp
    src/JuceBackend.cpp
)
target_link_libraries(heddle-render
    PRIVATE
        heddle_core
        juce::juce_core
        juce::juce_events
        juce::juce_audio_processors
        juce::juce_audio_formats
)
# NOTE: no juce_audio_plugin_client here either (same ODR reason as smoke).
target_compile_definitions(heddle-render PRIVATE JUCE_WEB_BROWSER=0)
target_compile_definitions(heddle-render PRIVATE JUCE_USE_CURL=0)
target_compile_definitions(heddle-render PRIVATE JUCE_PLUGINHOST_VST3=1)
# heddle_version.hpp is generated by configure_file() into the build dir.
target_include_directories(heddle-render PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
if(WIN32)
    target_link_libraries(heddle-render PRIVATE ws2_32 winmm)
endif()

# ---- heddle-host (the standalone live host) ---------------------------------
# heddle-host <chain.json> [--audio null|system] [--device NAME]
#            [--osc-port N] [--duration SECONDS] [--version]
#
# The full standalone app: loads the chain, runs the plugins live on a real
# output device (or a software null clock), and serves the OSC control
# protocol. This is the same process the future Ableton bridge VST3 routes
# into over localhost. It links juce_audio_devices (AudioDeviceManager) on
# top of the smoke/render module set — the host-side module JUCE's own
# AudioPluginHost uses for live IO. No juce_audio_plugin_client (same ODR
# reason as smoke/render: the host must not compile a second VST3 SDK base).
#
# juce_audio_devices is the only genuinely-new module here; its platform
# deps are already provisioned in CI (libasound2-dev on Linux — and the
# AudioDeviceManager sources are compiled into the target, so the X11/GUI
# stack CI installs is already sufficient).
add_executable(heddle-host
    src/host_main.cpp
    src/JuceBackend.cpp
)
target_link_libraries(heddle-host
    PRIVATE
        heddle_core
        juce::juce_core
        juce::juce_events
        juce::juce_graphics
        juce::juce_gui_basics
        juce::juce_audio_devices
        juce::juce_audio_processors
)
target_compile_definitions(heddle-host PRIVATE JUCE_WEB_BROWSER=0)
target_compile_definitions(heddle-host PRIVATE JUCE_USE_CURL=0)
target_compile_definitions(heddle-host PRIVATE JUCE_PLUGINHOST_VST3=1)
# JUCE 8 disables the modal dispatch loop by default; the headless host pumps
# its OSC Timer via MessageManager::runDispatchLoopUntil, which needs it on.
target_compile_definitions(heddle-host PRIVATE JUCE_MODAL_LOOPS_PERMITTED=1)
# heddle_version.hpp is generated by configure_file() into the build dir.
target_include_directories(heddle-host PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
if(WIN32)
    target_link_libraries(heddle-host PRIVATE ws2_32 winmm)
endif()

# ---- heddle-bridge-test (t5 bridge round-trip, headless) ----------------------
# heddle-bridge-test <heddle-host> <smoke-vst3>
#
# Drives the ring IPC end-to-end without loading the HeddleBridge VST3 (the
# plugin is a CI compile/instantiate check; this is the audio path). Spawns
# heddle-host headless with --audio ring, writes a 0.5 constant into
# heddle.bridge.in, reads heddle.bridge.out, and asserts the Gain-0.5 round-trip
# (mean ~= 0.25) plus an OSC /chain/query + /chain/quit round-trip.
#
# Links ONLY heddle_core (ring + OSC). No JUCE: no GUI, no audio device, no
# Xvfb -- it runs on a bare headless runner. On Windows the ws2_32 import
# needed by osc_io.cpp arrives transitively (heddle_core links it PUBLIC).
add_executable(heddle-bridge-test
    src/bridge_test.cpp
)
target_link_libraries(heddle-bridge-test
    PRIVATE
        heddle_core
)

# ---- heddle-osc-perf-host (OSC perf vs a REAL live host) -------------------
# heddle-osc-perf-host <heddle-host> <smoke-vst3> [--rates 50,200,500,2000,5000]
#
# The host-level half of the OSC performance story (see docs/PERFORMANCE.md).
# It spawns a REAL heddle-host (--audio ring, a live audio-thread pump + the
# smoke VST3) and, for a ramp of sustained OSC set-rates, measures:
#   * set->ack RTT (the realistic control latency through the message thread +
#     the live VST3 setParam, as a function of set rate);
#   * UDP set loss (max reliable rate, bounded by the 5 ms OSC drain + the
#     socket receive buffer);
#   * audio-path throughput deficit + input backlog (lock-contention cost: the
#     audio pump try-locks chainLock_, the OSC thread blocks on it).
# Like heddle-bridge-test it links ONLY heddle_core (no JUCE), so it runs on a
# bare headless runner; the child heddle-host does the JUCE/audio work.
add_executable(heddle-osc-perf-host
    src/osc_perf_host.cpp
)
target_link_libraries(heddle-osc-perf-host
    PRIVATE
        heddle_core
)

# ---- heddle-osc-loss-test (sample-level round-trip loss, headless) ----------
# heddle-osc-loss-test <heddle-host> <smoke-vst3> [--duration SEC]
#     [--rates 50,200,500,1000,2000,5000,10000] [--block N] [--rate SR] [--kick N]
#
# The SAMPLE-LEVEL complement to heddle-osc-perf-host. Instead of a block-count
# deficit it streams a known 4096-sample rectangular loop (1024 zero + 2048 x
# -1.5 + 1024 zero) into the host's input ring, captures the output ring, and
# verifies sample-for-sample that all 2048 non-zero samples per loop survive --
# baseline and under sustained OSC set load -- plus the round-trip offset and
# the control kick-in delay. Links ONLY heddle_core (no JUCE), runs headless;
# the child heddle-host does the JUCE/audio work.
add_executable(heddle-osc-loss-test
    src/osc_loss_test.cpp
)
target_link_libraries(heddle-osc-loss-test
    PRIVATE
        heddle_core
)
